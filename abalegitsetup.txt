local function getInput()
	local bp = LP:FindFirstChild("Backpack")
	if bp then
		local inp = bp:FindFirstChild("Input")
		if inp and inp:IsA("RemoteEvent") then return inp end
	end

	local char = LP.Character
	if char then
		local traits = char:FindFirstChild("ServerTraits")
		if traits then
			local inp = traits:FindFirstChild("Input")
			if inp and inp:IsA("RemoteEvent") then return inp end
		end
	end

	local bpTraits = bp and bp:FindFirstChild("ServerTraits")
	if bpTraits then
		local inp = bpTraits:FindFirstChild("Input")
		if inp and inp:IsA("RemoteEvent") then return inp end
	end

	for _, v in ipairs(LP:GetDescendants()) do
		if v:IsA("RemoteEvent") and v.Name == "Input" then
			return v
		end
	end

	return nil
end

local function isTeammate(model)
	if not State.farmTeamCheck then return false end
	if not LP.Team then return false end
	local plr = Players:GetPlayerFromCharacter(model)
	if not plr or not plr.Team then return false end
	return plr.Team == LP.Team
end

local function nearestTarget()
	local myChar = LP.Character
	local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
	if not myRoot then return nil end

	local best, bestDist = nil, math.huge
	for hum in pairs(hooked) do
		local model = hum.Parent
		if model and model.Parent and model ~= myChar
			and not isTeammate(model) and hum.Health > 0 then
			local root = model:FindFirstChild("HumanoidRootPart")
			if root then
				local dist = (myRoot.Position - root.Position).Magnitude
				if dist < bestDist then
					best, bestDist = root, dist
				end
			end
		end
	end
	return best
end

local function virtualClick()
	local cam = workspace.CurrentCamera
	local vp = cam and cam.ViewportSize or Vector2.new(800, 600)
	local x = math.floor(vp.X * 0.5)
	local y = math.floor(vp.Y * 0.5)
	VirtualInput:SendMouseButtonEvent(x, y, 0, true, game, 0)
	task.wait(0.02)
	VirtualInput:SendMouseButtonEvent(x, y, 0, false, game, 0)
end

local function fireM1()
	local input = getInput()
	if not input then return end

	local char = LP.Character
	local root = char and char:FindFirstChild("HumanoidRootPart")
	local cam = workspace.CurrentCamera
	local target = nearestTarget()

	local hitPos, lookDir
	if target then
		hitPos = target.Position
		lookDir = target.Position - (root and root.Position or cam.CFrame.Position)
		lookDir = lookDir.Magnitude > 0 and lookDir.Unit or cam.CFrame.LookVector
	elseif cam then
		local origin = root and root.Position or cam.CFrame.Position
		lookDir = cam.CFrame.LookVector
		hitPos = origin + lookDir * 10
	else
		hitPos = root and (root.Position + root.CFrame.LookVector * 10) or Vector3.new()
		lookDir = root and root.CFrame.LookVector or Vector3.new(0, 0, -1)
	end

	local mouseHit = CFrame.new(hitPos, hitPos + lookDir)
	local down = { "M1", { air = false, md = Vector3.new(), mousehit = mouseHit, skeydown = true, skeyreal = false } }
	local up   = { "M1", { air = false, md = Vector3.new(), mousehit = mouseHit, skeydown = false, skeyreal = true } }

	pcall(function() input:FireServer(unpack(down)) end)
	task.wait(0.05)
	pcall(function() input:FireServer(unpack(up)) end)
end

local function stopHeadRide()
	State.headRideTarget = nil
	State.headRideEnd = 0
	State.headRidePending = false
	if State.headRideConn then
		pcall(function() State.headRideConn:Disconnect() end)
		State.headRideConn = nil
	end
end

local function startHeadRide(hum)
	if not State.headRide then return end
	if State.headRideTarget == hum then return end

	State.headRideTarget = hum
	State.headRideEnd = tick() + State.headRideDur

	if State.headRideConn then
		State.headRideConn:Disconnect()
		State.headRideConn = nil
	end

	State.headRideConn = RunService.RenderStepped:Connect(function(dt)
		if not State.headRide or tick() >= State.headRideEnd then
			stopHeadRide()
			return
		end

		local model = hum and hum.Parent
		if not model or not model.Parent then stopHeadRide() return end

		local myChar = LP.Character
		local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
		local theirHead = model:FindFirstChild("Head")
		if not myRoot or not theirHead then stopHeadRide() return end

		if theirHead.AssemblyLinearVelocity.Y < -60 then stopHeadRide() return end

		local targetCF = theirHead.CFrame * CFrame.new(0, State.headRideOffset, 0)
		local alpha = 1 - math.exp(-dt / math.max(State.headRideAlpha, 1e-4))
		myRoot.CFrame = myRoot.CFrame:Lerp(targetCF, alpha)
		myRoot.AssemblyLinearVelocity = Vector3.new()
	end)
end

local function waitForHitThenRide()
	if not State.headRide or State.headRidePending then return end
	State.headRidePending = true

	local char = LP.Character
	local myRoot = char and char:FindFirstChild("HumanoidRootPart")
	if not myRoot then
		State.headRidePending = false
		return
	end

	local best, bestDist = nil, math.huge
	for hum in pairs(hooked) do
		local model = hum.Parent
		if model and model.Parent and model ~= char then
			local root = model:FindFirstChild("HumanoidRootPart")
			if root then
				local dist = (root.Position - myRoot.Position).Magnitude
				if dist < bestDist then
					best, bestDist = hum, dist
				end
			end
		end
	end

	if not best then
		State.headRidePending = false
		return
	end

	local targetModel = best.Parent
	local startHealth = best.Health
	local startTags = {}
	for _, c in ipairs(targetModel:GetChildren()) do
		startTags[c.Name] = true
	end

	local startTime = tick()
	local conn
	conn = RunService.Heartbeat:Connect(function()
		if not State.headRide or tick() - startTime > 0.6 then
			conn:Disconnect()
			State.headRidePending = false
			return
		end

		local hum = targetModel:FindFirstChildOfClass("Humanoid")
		if not hum then
			conn:Disconnect()
			State.headRidePending = false
			return
		end

		local hit = hum.Health < startHealth
		if not hit then
			for _, c in ipairs(targetModel:GetChildren()) do
				if not startTags[c.Name] then
					local n = c.Name:lower()
					if n:find("hit") or n:find("stun") or n:find("slam")
						or n:find("ragdoll") or n:find("got") then
						hit = true
						break
					end
				end
			end
		end

		if hit then
			conn:Disconnect()
			State.headRidePending = false
			startHeadRide(hum)
		end
	end)
end

local function hookMyAnimator()
	if State.myAnimConn then
		pcall(function() State.myAnimConn:Disconnect() end)
		State.myAnimConn = nil
	end

	local char = LP.Character

		if not State.headRide then return end
		local id = track.Animation and track.Animation.AnimationId
		if id == HEAD_RIDE_ANIM then
			waitForHitThenRide()
		end
	end)
end

local function blockPulse(duration)
	local input = getInput()
	if not input then return end

	local now = tick()
	local newEnd = now + duration

	if State.blockEndThread and State.blockHoldUntil > now and newEnd <= State.blockHoldUntil then
		return
	end

	if State.blockEndThread then
		pcall(task.cancel, State.blockEndThread)
		State.blockEndThread = nil
	end

	pcall(function() input:FireServer("BlockOn") end)
	State.blockHoldUntil = newEnd

	State.blockEndThread = task.delay(duration, function()
		local i = getInput()
		if i then pcall(function() i:FireServer("BlockOff") end) end
		State.blockEndThread = nil
		State.blockHoldUntil = 0
	end)
end

local function onAttack(animId)
	if not State.autoBlock and not State.trade then return end

	local timing       = animId and animTimings[animId] or nil
	local hasBlockTime = timing and timing.block ~= nil
	local hasTradeTime = timing and timing.trade ~= nil
	local ping         = State.pingComp and State.pingCompMs / 1000 or 0

	if State.autoBlock then
		if hasBlockTime then
			blockPulse(timing.block + ping)
		else
			blockPulse(State.blockHold + ping)
		end
	end

	if State.trade then
		local now = tick()
		if now - State.lastTradeAt < State.tradeWindow then return end
		State.lastTradeAt = now

		task.spawn(function()
			local waitTime = hasTradeTime
				and (timing.trade + State.tradeDelay)
				or  (State.blockHold + ping + State.tradeDelay)
			task.wait(waitTime)

			for i = 1, State.tradeHits do
				if not State.trade then break end
				fireM1()
				if i < State.tradeHits then task.wait(State.tradeHitGap) end
			end
		end)
	end
end

local function farmPositionCFrame(targetRoot)
	local pos  = targetRoot.Position
	local look = targetRoot.CFrame.LookVector

	if State.farmPosMode == "On Top"   then return CFrame.new(pos + Vector3.new(0, 4, 0)) end
	if State.farmPosMode == "Below"    then return CFrame.new(pos - Vector3.new(0, 4, 0)) end
	if State.farmPosMode == "Behind"   then return CFrame.new(pos - look * 4, pos) end
	if State.farmPosMode == "In Front" then return CFrame.new(pos + look * 4, pos) end
	return CFrame.new(pos)
end

local function farmLoop()
	if State.farmConn then State.farmConn:Disconnect() State.farmConn = nil end
	if State.farmM1Thread then task.cancel(State.farmM1Thread) State.farmM1Thread = nil end
	if State.farmSkillThread then task.cancel(State.farmSkillThread) State.farmSkillThread = nil end

	State.farmRotIndex = 1

	State.farmConn = RunService.Heartbeat:Connect(function()
		if not State.farm then
			if State.farmConn then State.farmConn:Disconnect() State.farmConn = nil end
			return
		end

		local char = LP.Character
		local myRoot = char and char:FindFirstChild("HumanoidRootPart")
		if not myRoot then return end

		local best, bestDist = nil, State.farmRange
		for hum in pairs(hooked) do
			local model = hum.Parent
			if model and model.Parent and model ~= char and not isTeammate(model) then
				local root = model:FindFirstChild("HumanoidRootPart")
				local h    = model:FindFirstChildOfClass("Humanoid")
				if root and h and h.Health > 0 then
					local dist = (root.Position - myRoot.Position).Magnitude
					if dist < bestDist then
						best, bestDist = hum, dist
					end
				end
			end
		end

		if not best then
			State.farmTarget = nil
			State.forcedLock = nil
			return
		end

		State.farmTarget = best
		local targetRoot = best.Parent:FindFirstChild("HumanoidRootPart")
		if not targetRoot then return end

		State.forcedLock = targetRoot
		myRoot.CFrame = farmPositionCFrame(targetRoot)
		myRoot.AssemblyLinearVelocity = Vector3.new()
	end)

	State.farmM1Thread = task.spawn(function()
		while State.farm do
			if State.farmTarget and State.farmTarget.Parent then
				fireM1()
				task.wait(State.farmM1Delay)
			else
				task.wait(0.05)
			end
		end
	end)

	State.farmSkillThread = task.spawn(function()
		while State.farm do
			if State.farmTarget and State.farmTarget.Parent then
				fireNextFarmSkill()
			end
			task.wait(State.farmSkillDelay)
		end
	end)
end

local function hookHumanoid(hum)
	if not hum or not hum:IsA("Humanoid") or hooked[hum] then return end

	local animator = hum:FindFirstChildOfClass("Animator")
	if not animator then
		task.spawn(function()
			local waited = 0
			while hum.Parent and not animator and waited < 5 do
				task.wait(0.1)
				waited += 0.1
				animator = hum:FindFirstChildOfClass("Animator")
			end
			if animator then hookHumanoid(hum) end
		end)
		return
	end

	hooked[hum] = true
	local model = hum.Parent

	local conn = animator.AnimationPlayed:Connect(function(track)
		if not model or not model.Parent then return end

		local id = track.Animation and track.Animation.AnimationId
		if not id then return end
		if comboEndIds[id] then return end
		if not atkIds[id] and not animTimings[id] then return end
		if not State.autoBlock and not State.trade then return end

		local myChar = LP.Character
		local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
		local theirRoot = model:FindFirstChild("HumanoidRootPart")
		if not myRoot or not theirRoot then return end
		if (myRoot.Position - theirRoot.Position).Magnitude > State.blockRange then return end

		onAttack(id)
	end)

	local ancestryConn
	ancestryConn = hum.AncestryChanged:Connect(function(_, parent)
		if parent then return end
		pcall(function() conn:Disconnect() end)
		pcall(function() ancestryConn:Disconnect() end)
		hooked[hum] = nil
	end)

	hum.ChildAdded:Connect(function(c)
		if c:IsA("Animator") then
			hooked[hum] = nil
			hookHumanoid(hum)
		end
	end)

	hum.ChildRemoved:Connect(function(c)
		if c:IsA("Animator") then
			hooked[hum] = nil
			hookHumanoid(hum)
		end
	end)
end

		end
	end
end

local function hookPlayersFunc()

	local myChar = LP.Character

			local c = plr.Character
			if c and c ~= myChar then
				local h = c:FindFirstChildOfClass("Humanoid")
				if h then hookHumanoid(h) end
			end
		end
	end
end

local function scanAll()
	local myChar = LP.Character
	for _, o in ipairs(workspace:GetDescendants()) do
		if o:IsA("Humanoid") and not (o.Parent and myChar and o.Parent == myChar) then
			hookHumanoid(o)
		end
	end
end

local HookConns = {}
local function startHooks()
	hookLive()
	hookPlayersFunc()
	scanAll()

	local live = workspace:FindFirstChild("Live")
	if live then
		HookConns.live = live.ChildAdded:Connect(function(c)
			if not State.autoBlock and not State.trade and not State.headRide and not State.farm then return end
			if not State.hookDummies then return end
			if c:IsA("Model") then
				local h = c:FindFirstChildOfClass("Humanoid")
				if h then hookHumanoid(h) end
			end
		end)
	end

	HookConns.descendant = workspace.DescendantAdded:Connect(function(o)
		if not State.autoBlock and not State.trade and not State.headRide and not State.farm then return end
		if o:IsA("Humanoid") then
			local mc = LP.Character
			if not (o.Parent and mc and o.Parent == mc) then hookHumanoid(o) end
		elseif o:IsA("Animator") then
			local h = o.Parent
			if h and h:IsA("Humanoid") then hookHumanoid(h) end
		end
	end)

	HookConns.playerAdded = Players.PlayerAdded:Connect(function(plr)

local cachedBar, cachedBarOwner
local function findBar()
	local char = LP.Character
	if cachedBar and cachedBarOwner == char and cachedBar.Parent then
		local cut  = cachedBar:FindFirstChild("Cutter")
		local goal = cachedBar:FindFirstChild("Goal")
		if cut and goal then return cut, goal end
	end

	local live = workspace:FindFirstChild("Live")
	if not live then return nil end

	for _, c in ipairs(live:GetChildren()) do
		local hrp = c:FindFirstChild("HumanoidRootPart")
		local gui = hrp and hrp:FindFirstChild("NanamiCutGUI")
		local bar = gui and gui:FindFirstChild("MainBar")
		if bar then
			local cut  = bar:FindFirstChild("Cutter")
			local goal = bar:FindFirstChild("Goal")
			if cut and goal then
				cachedBar = bar
				cachedBarOwner = char
				return cut, goal
			end
		end
	end

	return nil
end

local function nanamiTick()
	if not State.nanami or State.nanamiFired then return end

	local char = LP.Character
	if not char or char:GetAttribute("NANAMIAIM") ~= true then return end

	local cut, goal = findBar()
	if not cut or not goal then return end

	if math.abs(cut.Position.X.Scale - goal.Position.X.Scale) <= State.nanamiTol then
		virtualClick()
		State.nanamiFired = true
	end
end

local function yujiTick()
	if not State.yuji then return end

	local cam = workspace.CurrentCamera
	if not cam then return end

	local fov = cam.FieldOfView
	if fov < State.yujiEnter then State.yujiChanged = true end

	if State.yujiChanged and not State.yujiFired and fov >= State.yujiThreshold then
		fireSkillByMode()
		State.yujiFired = true
	end

	if fov > State.yujiReset or fov < State.yujiEnter - 5 then
		State.yujiChanged = false
		State.yujiFired = false
	end
end

		State.yujiKeyConn = nil
	end

	State.yujiKeyConn = UserInputService.InputBegan:Connect(function(input, gp)

		if input.KeyCode ~= State.yujiKey then return end
		task.spawn(function()
			task.wait(State.yujiKeyDelay)
			if State.yuji then fireSkillByMode() end
		end)

end

local function lockTick()
	local target = State.forcedLock

	if not target then
		if not State.lock then return end

		local myChar = LP.Character
		local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
		if not myRoot then return end

		local best, bestDist = nil, State.lockRange
		for hum in pairs(hooked) do
			local model = hum.Parent
			if model and model.Parent and model ~= myChar and not isTeammate(model) then
				local root = model:FindFirstChild("HumanoidRootPart")
				if root then
					local dist = (root.Position - myRoot.Position).Magnitude
					if dist < bestDist then
						best, bestDist = root, dist
					end
				end
			end
		end
		target = best
	end

	if not target then return end

	local myChar = LP.Character
	local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
	local hum = myChar and myChar:FindFirstChildOfClass("Humanoid")
	if not myRoot or not hum then return end

	local p = myRoot.Position
	local t = target.Position
	if hum.AutoRotate then hum.AutoRotate = false end
	myRoot.CFrame = CFrame.lookAt(p, Vector3.new(t.X, p.Y, t.Z))
end

local function doRejoin()
	pcall(function() TeleportService:Teleport(game.PlaceId, LP) end)
end

local function doServerHop()
	task.spawn(function()
		local okHop = pcall(function()
			local url = "https://games.roblox.com/v1/games/" .. game.PlaceId
				.. "/servers/Public?sortOrder=Asc&limit=100"
			local data = HttpService:JSONDecode(game:HttpGet(url))
			if data and data.data then
				for _, srv in ipairs(data.data) do
					if srv.playing and srv.maxPlayers and srv.playing < srv.maxPlayers
						and srv.id ~= game.JobId then
						TeleportService:TeleportToPlaceInstance(game.PlaceId, srv.id, LP)
						return
					end
				end
			end
			TeleportService:Teleport(game.PlaceId, LP)
		end)
		if not okHop then doRejoin() end
	end)
end

local function doJoinLowest()
	task.spawn(function()
		local okJoin = pcall(function()
			local cursor = ""
			local bestId, bestCount = nil, math.huge

			for _ = 1, 3 do
				local url = "https://games.roblox.com/v1/games/" .. game.PlaceId
					.. "/servers/Public?sortOrder=Asc&limit=100"
				if cursor ~= "" then url = url .. "&cursor=" .. cursor end

				local data = HttpService:JSONDecode(game:HttpGet(url))
				if not (data and data.data) then break end

				for _, srv in ipairs(data.data) do
					if srv.playing and srv.maxPlayers and srv.playing < srv.maxPlayers
						and srv.id ~= game.JobId and srv.playing < bestCount then
						bestCount = srv.playing
						bestId = srv.id
					end
				end

				cursor = data.nextPageCursor or ""
				if cursor == "" then break end
			end

			if bestId then
				TeleportService:TeleportToPlaceInstance(game.PlaceId, bestId, LP)
			else
				TeleportService:Teleport(game.PlaceId, LP)
			end		end)
		if not okJoin then doRejoin() end
	end)
end

local function hookLog(hum)
	if not hum or not hum:IsA("Humanoid") or logHooked[hum] then return end

	local animator = hum:FindFirstChildOfClass("Animator")
	if not animator then
		task.spawn(function()
			local waited = 0
			while hum.Parent and not animator and waited < 5 do
				task.wait(0.1)
				waited += 0.1
				animator = hum:FindFirstChildOfClass("Animator")
			end
			if animator then hookLog(hum) end
		end)
		return
	end

	logHooked[hum] = true
	local model = hum.Parent

	animator.AnimationPlayed:Connect(function(track)
		local anim = track.Animation
		if not anim then return end
		local id = anim.AnimationId
		if not id or id == "" then return end
		if model == LP.Character then return end
		if LOCO_NAMES[anim.Name] then return end

		local myRoot    = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
		local theirRoot = model:FindFirstChild("HumanoidRootPart")
		if not myRoot or not theirRoot then return end

		local d = (myRoot.Position - theirRoot.Position).Magnitude
		if d <= Logger.range then
			task.spawn(addLogRow, id, anim.Name, model.Name, math.floor(d))
		end
	end)
end

local function hookMyLogger(hum)
	if not hum or not hum:IsA("Humanoid") or myLogHooked[hum] then return end

	local animator = hum:FindFirstChildOfClass("Animator")
	if not animator then
		task.spawn(function()
			local waited = 0
			while hum.Parent and not animator and waited < 5 do
				task.wait(0.1)
				waited += 0.1
				animator = hum:FindFirstChildOfClass("Animator")
			end
			if animator then hookMyLogger(hum) end
		end)
		return
	end

	myLogHooked[hum] = true

	animator.AnimationPlayed:Connect(function(track)
		local anim = track.Animation
		if not anim then return end
		local id = anim.AnimationId
		if not id or id == "" then return end

		if getgenv().OutlawIncludeLoco then
			task.spawn(addMyLogRow, id, anim.Name)
		elseif getgenv().OutlawFilterLoco ~= false and not isLocomotion(id, anim.Name) then
			task.spawn(addMyLogRow, id, anim.Name)
		end
	end)
end

local function bindMyLoggerChar(char)
	if not char then return end

	local hum = char:FindFirstChildOfClass("Humanoid")
	if hum then hookMyLogger(hum) end

	char.ChildAdded:Connect(function(c)
		if c:IsA("Humanoid") then hookMyLogger(c) end
	end)
	char.DescendantAdded:Connect(function(c)
		if c:IsA("Humanoid") then hookMyLogger(c) end
	end)
end

local function hookContainer(container)
	if not container then return end
	for _, c in ipairs(container:GetChildren()) do
		if c:IsA("Model") then
			local h = c:FindFirstChildOfClass("Humanoid")
			if h then hookLog(h) end
		end
	end
end

		local id = Visualizer.currentId
		if not id then

			tradeMarker.Visible = false

			refreshBadge()
			return
		end

		local t   = animTimings[id]
		local dur = Visualizer.duration

		if t and t.block and dur > 0 then
			blockMarker.Visible = true
			blockMarker.Position = UDim2.new(t.block / dur, 0, 0, -3)
		else
			blockMarker.Visible = false
		end

		if t and t.trade and dur > 0 then
			tradeMarker.Visible = true
			tradeMarker.Position = UDim2.new(t.trade / dur, 0, 0, -3)
		else
			tradeMarker.Visible = false
		end

		local bs = t and t.block and string.format("%.3fs", t.block) or "--"
		local ts = t and t.trade and string.format("%.3fs", t.trade) or "--"
		timingLbl.Text = "block: " .. bs .. " | trade: " .. ts

	end

			pcall(function() Visualizer.track.TimePosition = Visualizer.pausedTime end)
			pcall(function() Visualizer.track:AdjustSpeed(Visualizer.speed) end)
